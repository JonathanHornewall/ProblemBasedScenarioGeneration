\section*{Impact and visibility of related work (as of Sept.\ 2025)}
\paragraph{Tier A --- Highly cited ($\approx 500+$).}
\begin{itemize}
  \item \citet{elmachtoub2021spo} --- venue pages report $\sim$1000 citations; canonical DFL surrogate for linear objectives.
  \item \citet{bertsimas2020predictive} --- $\sim$1000 citations; cornerstone for contextual/prescriptive analytics.
  \item \citet{ban2019bigdata} --- $\sim$670 citations (well above the 500+ threshold).
  \item \citet{donti2017task} --- venue trackers show $\sim$500 citations; widely regarded as an early DFL-in-SP reference.
\end{itemize}

\paragraph{Tier B --- Established building blocks ($\approx 100$--$500$).}
\begin{itemize}
  \item \citet{kallus2023sof} --- $\sim$90 citations; near the 100 mark and growing.
\end{itemize}

\paragraph{Tier C --- Solid mid-field ($\approx 20$--$100$).}
\begin{itemize}
  \item \citet{munoz2022bilevel} --- $\sim$60 citations; widely used in energy/operations case studies.
  \item \citet{stratigakos2022prescriptive} --- $\sim$70+ citations; visible in the power systems community.
  \item \citet{deng2022predictive} --- $\sim$27 citations; conceptual anchor for predictive SP framing.
  \item \citet{kannan2021hetero} --- $\sim$20+ citations; contextual residuals line.
\end{itemize}

\paragraph{Tier D --- Recent/emerging (new OR publications or preprints; citations not yet indicative).}
\begin{itemize}
  \item \citet{estes2023twostage} --- $\sim$10+ citations; journal version is 2023.
  \item \citet{zhu2024residualcdro} --- arXiv shows single-digit citations ($\sim$3).
  \item \citet{diasgarcia2024application} --- very recent; impact will lag indexing.
  \item \citet{islip2025contextual} --- new; impact to depend on scale/ablations and code release.
  \item \citet{wu2022cvae} --- known in DFL-for-MILPs via talks/code; formal counts not consolidated.
  \item \citet{persak2024dff} --- first multistage DFL with state-path gradients; brand-new, counts not yet meaningful.
  \item \citet{iglesias2025prescribeselect} --- proposes adaptive meta-policy selection across contextual methods; very recent.
\end{itemize}

\paragraph{Notes on methodology.} We rely on publisher venue pages (INFORMS/ACM/NeurIPS) and IDEAS/RePEc when available to dampen index-to-index variance. Google Scholar typically reports higher counts due to broader coverage; venue pages provide a conservative baseline. Because citation accrual is dynamic, these tiers should be read as order-of-magnitude signals, not exact rankings.

\paragraph{Takeaway.} The most recognized anchors remain \citet{elmachtoub2021spo,bertsimas2020predictive,ban2019bigdata,donti2017task}. Everything else should be positioned against these Tier~A references, with Tier~B/C works supplying relevant contextual and decision-focused comparators (e.g., \citet{kallus2023sof,munoz2022bilevel,deng2022predictive}). Newer DFL-for-scenarios (\citet{islip2025contextual,wu2022cvae}) and multistage DFL (\citet{persak2024dff}) are emerging—credible, but their formal impact will be established by large-scale, reproducible evaluations over the next cycles.
