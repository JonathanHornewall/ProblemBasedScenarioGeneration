\begin{abstract}
Decision focused scenario learning for contextual two-stage stochastic programming has emerged as a promising research direction. But challenges exist.
A general theory for when such approaches can theoretically yield optimal decision policies is currently lacking. 
Additionally, existing methods typically require specific assumptions on which second stage data is random. 
Generally, the non-differentiability of the decision-focused objective is a source of difficulty. 
This paper makes three theoretical contributions, each aiming to remedy one of these challenges.
We present the first comprehensive theory for when scenario generators can induce optimal policies from a single generated scenario.
Second we demonstrate a theoretical and methodological equivalence result between learning different second-stage data components, significantly extending existing methods to instances beyond the regimes they were designed for. 
Third, we study log-barrier smoothing, showing how it enables training by inducing smoothness and componentwise invexity in the scenario parametrized decision loss. 
Leveraging the theory, we present a novel method inspired by interior point methods from classical optimization. It has the benefit of learning directly on the downstream decision objective without the need for surrogates, and has a natural selection rule for when optima are non-unique. 
We validate our method and theory on standard benchmarks.
\end{abstract}

\section{Notation and Problem Statement}
\label{sec:notation-problem}

\subsection{Data set and scenarios}
\label{sec:data-scenarios}

A context variable $x\in \R^{n_x}$ is represented as a Euclidean vector. A scenario is represented as a collection of three \emph{scenario components}
\[
\xi=\bigl(W(\xi),h(\xi),q(\xi)\bigr)
\in \R^{m_2\times n_2}\times\R^{m_2}\times\R^{n_2},
\]
where the component $W(\xi)\in \R^{m_2\times n_2}$ represents a second-stage constraint matrix (recourse matrix), the component $h(\xi)\in \R^{m_2}$ represents a second-stage constraint vector (demand), and the component $q(\xi)\in \R^{n_2}$ represents a second-stage cost vector (cost).

We assume the context and scenarios are sampled from a set $\mathcal X \times \Xi$ according to an unknown joint distribution $P$, and that we have access to a dataset $\mathcal D$ of $N$ context-scenario pairs
\begin{equation}
\label{eq:dataset}
\mathcal D:=\{(x_i,\xi_i)\}_{i=1}^N\subseteq \mathcal X\times \Xi,
\qquad
(x_i,\xi_i)\overset{\mathrm{i.i.d.}}{\sim}P.
\end{equation}

Throughout, $A\in\R^{m_1\times n_1}$, $b\in\R^{m_1}$, $c\in\R^{n_1}$, and $T\in\R^{m_2\times n_1}$ denote deterministic first-stage and transition data.

\subsection{Contextual stochastic programming with log-barrier smoothing}
\label{sec:contextual-logbarrier}

For each $x\in \mathcal X$, and each admissible choice of log-barrier smoothing parameter $\mu \in [0,\infty)$, the log-barrier smoothed two-stage stochastic linear program is given by
\begin{equation}
\label{eq:Pmu-x}
\begin{aligned}
(\mathcal{P}_\mu^{x})\qquad
&\min_{z\in Z}
\left\{
c^\top z-\mu\sum_{j=1}^{n_1}\log z_j
+
\E_{\boldsymbol{\xi}\sim P(\cdot\mid x)}\!\left[Q_\mu(z,\boldsymbol{\xi})\right]
\right\},
\\
&Z:=\{z\in\R^{n_1}_+:\;Az=b\}.
\end{aligned}
\end{equation}
where $Q_\mu$ denotes the second-stage pay-off
\begin{equation}
\label{eq:Qmu}
\begin{aligned}
Q_\mu(z,\xi)
&:=
\min_{y\in\mathcal Y(z,\xi)}
q(\xi)^\top y-\mu\sum_{k=1}^{n_2}\log y_k,
\\
\mathcal Y(z,\xi)
&:=
\left\{
y\in\R^{n_2}_+:\; W(\xi)y=h(\xi)-Tz
\right\}.
\end{aligned}
\end{equation}
Our main goal is to solve the unsmoothed problem, obtained by setting $\mu = 0$. We use the log-barrier smoothing as a means to this end.

We make the following assumptions:
\begin{itemize}
    \item $\mathcal X$ and $\Xi$ are compact, and $P$ has finite first moment (statistical regularity).
    \item For every $x\in\mathcal X$, $Z$ is nonempty and has nonempty relative interior (first-stage feasibility), and $\mathcal Y(z,\xi)$ has nonempty relative interior for every $z\in Z$ and every $\xi$ in the conditional support of $P(\cdot\mid x)$ (recourse feasibility).
    \item For every $x\in\mathcal X$, the problem $\mathcal{P}_\mu^{x}$ has a solution for every value of $\mu\geq 0$ (well-posedness).
\end{itemize}
None of these assumptions are invasive, and any well-posed problem can be reformulated to respect them.

Throughout, we will use the following equivalent formulation of $\mathcal{P}_\mu^{x}$:
\begin{equation}
\label{eq:Gmu-min}
\min_{z\in\R^{n_1}} G_\mu(z,x).
\end{equation}
Where
\begin{equation}
\label{eq:Gmu}
G_\mu(z,x):=\E_{\boldsymbol{\xi}\sim P(\cdot\mid x)}\!\left[\hat G_\mu(z,\boldsymbol{\xi})\right].
\end{equation}
With $\hat G_\mu$ denoting the scenario-specific cost function given by
\begin{equation}
\label{eq:hatGmu}
\hat G_\mu(z,\xi):=
c^\top z
-\mu\sum_{j=1}^{n_1}\log z_j
+Q_\mu(z,\xi)
+\iota_{Z}(z),
\end{equation}
where $\iota_{Z}(z)=0$ if $z\in Z$ and $\iota_{Z}(z)=+\infty$ otherwise.

For the original $\mu = 0$ problem, $\mathcal{P}_\mu^{x}$ may have multiple optimal solutions. However, for the smoothed problem with $\mu > 0$, the solution is unique. For $\mu \geq 0$, we denote the unique optimal policy by
\begin{equation}
\label{eq:zsharp-mu}
z_\mu^\#:\mathcal X\to Z,
\qquad
z_\mu^\#(x):=\argmin_{z\in\R^{n_1}}G_\mu(z,x),
\end{equation}
with the case $\mu=0$ defined by the log-barrier central-path limit below.

We will frequently omit the $\mu$ subscript when considering the unsmoothed, $\mu = 0$ case. By classical theory, the problem returns to its original non-smoothed version as $\mu$ goes to zero (see appendix for an elaborate explanation):
\begin{equation}
\label{eq:G-limit}
G(z,x):=\lim_{\mu\downarrow 0}G_\mu(z,x),
\qquad
\hat G(z,\xi):=\lim_{\mu\downarrow 0}\hat G_\mu(z,\xi).
\end{equation}
In particular, we define
\begin{equation}
\label{eq:zsharp-limit}
z^\#(x)=\lim_{\mu \downarrow 0} z_\mu^\#(x) \in \argmin_{z\in Z} G(z,x)
\end{equation}
as the unique optimal policy over the unsmoothed problem, defined by the log-barrier smooth limit.

\subsection{Synthetic scenarios}
\label{sec:synthetic-scenarios}

Throughout, we will frequently consider ``synthetic'' feasible scenarios
\begin{equation}
\label{eq:Xi-synthetic}
\Xi^{\mathrm{synthetic}}:=
\left\{
(W,h,q)\in
\R^{m_2\times n_2}\times\R^{m_2}\times\R^{n_2}:
-\infty<
\min_{z\in Z}\hat G(z,(W,h,q))
<\infty
\right\}.
\end{equation}
Intuitively, $\Xi^{\mathrm{synthetic}}$ represents all scenarios that result in a feasible and bounded single-scenario problem. They need not be included in the support of the randomness $\Xi$, and may not have a physical interpretation. In such cases, they are best interpreted as a decision parameterization, rather than an actual forecast.

We define the optimal decision map by:
\begin{equation}
\label{eq:single-scenario-map-mu}
\hat z_\mu^\#:\Xi^{\mathrm{synthetic}}\to Z,
\qquad
\hat z_\mu^\#(\xi):=\argmin_{z\in\R^{n_1}}\hat G_\mu(z,\xi),
\end{equation}
\begin{equation}
\label{eq:single-scenario-map}
\hat z^\#:\Xi^{\mathrm{synthetic}}\to Z,
\qquad
\hat z^\#(\xi):=\lim_{\mu\downarrow 0}\hat z_\mu^\#(\xi)\in\argmin_{z\in Z}\hat G(z,\xi).
\end{equation}

The decision focused loss function over scenarios is
\begin{equation}
\label{eq:L}
L(\xi,x):=G(\hat z^\#(\xi),x)
=\E_{\boldsymbol{\xi}'\sim P(\cdot\mid x)}\!\left[\hat L(\xi,\boldsymbol{\xi}')\right],
\end{equation}
where
\begin{equation}
\label{eq:hatL}
\hat L(\xi,\xi'):=\hat G(\hat z^\#(\xi),\xi')
\end{equation}
is the scenario-specific loss, interpreted as the cost of scenario $\xi'$ after optimizing with respect to scenario $\xi$.

\subsection{Decision focused scenario generation}
\label{sec:decision-focused-scenario-generation}

As a learning problem, our contextual stochastic optimization reduces to solving the empirical minimization problem defined by
\begin{equation}
\label{eq:direct-policy-erm}
\min_{\eta}\ \frac1N\sum_{i=1}^N G(\pi_\eta(x_i),x_i),
\qquad
\pi_\eta:\mathcal X\to Z,
\end{equation}
where $\eta$ parametrizes the optimization family.
The decision focused scenario generation approach specifically consists of solving the problem
\begin{equation}
\label{eq:scenario-generation-erm}
\min_{w}\ \frac1N\sum_{i=1}^N
\hat G\!\left(\hat z^\#(\phi_w(x_i)),\xi_i\right),
\qquad
\phi_w:\mathcal X\to \Xi^{\mathrm{synthetic}}.
\end{equation}
We remark that decision focused scenario generation can be viewed as a specific case of direct policy learning, by setting the policy family in \eqref{eq:direct-policy-erm} as equal to all policies obtained by composing $\hat z^{\#}$ with the policies in the scenario policy class:
\begin{equation}
\label{eq:scenario-policy-class}
\Pi_{\mathrm{SG}}:=
\left\{
\pi_w:\mathcal X\to Z:\;
\pi_w=\hat z^\#\circ \phi_w,\ \phi_w:\mathcal X\to \Xi^{\mathrm{synthetic}}
\right\}.
\end{equation}

\section{Optimal policy realizability via scenario generation}
\label{sec:optimal-policy-realizability}

\begin{theorem}[Optimal policy realizability via scenario generation]
\label{thm:realizability}
Assume a deterministic technology matrix $\forall \xi\in\Xi$, $T(\xi)=T_{\det}$. Fix a context $x\in\mathcal X$ and a scenario $\xi\in\Xi$. Let $\mu \in [0,\infty)$ denote any admissible choice of log-barrier smoothing parameter, including $0$.
Define
\begin{equation}
\label{eq:qepsilon}
q_\epsilon(\xi)[i]:=
\begin{cases}
q(\xi)[i], & q(\xi)[i]\neq 0,\\
\epsilon, & q(\xi)[i]=0,
\end{cases}
\qquad i=1,\ldots,n_2,\quad \epsilon>0,
\end{equation}
and
\begin{equation}
\label{eq:qmaxdelta}
q_\delta^{\max}[i]:=\max_{\xi'\in\Xi}\left(q(\xi')[i]+\delta\right),
\qquad i=1,\ldots,n_2,\quad \delta>0.
\end{equation}
There exist decision-optimal components
$q_\mu^\#(x,\xi)\in\R^{n_2}$ and $W_{\mu,\epsilon}^\#(x,\xi)\in\R^{m_2\times n_2}$ in the sense that
\begin{equation}
\label{eq:q-decision-optimal}
z_\mu^\#(x)
=
\hat z_\mu^\#\!\left(
W(\xi),h(\xi),q_\mu^\#(x,\xi)
\right),
\end{equation}
\begin{equation}
\label{eq:W-decision-optimal}
z_\mu^\#(x)
=
\hat z_\mu^\#\!\left(
W_{\mu,\epsilon}^\#(x,\xi),h(\xi),q_\epsilon(\xi)
\right),
\end{equation}
If the recourse is deterministic $\forall \xi\in\Xi$, $W(\xi)=W_{\det}$, there additionally exists a decision optimal
$h_{\mu,\delta}^\#(x)\in\R^{m_2}$ such that
\begin{equation}
\label{eq:h-decision-optimal}
z_\mu^\#(x)
=
\hat z_\mu^\#\!\left(
W_{\det},h_{\mu,\delta}^\#(x),q_\delta^{\max}
\right),
\end{equation}
\end{theorem}

\begin{remark}[Synthetic scenarios]
\label{rem:synthetic-scenarios}
While the theorem may seem magic at first, we emphasize that it does not guarantee that the decision optimal scenario will exist in the support of the distribution of $\xi$. The scenario is ``synthetic'', in the sense that it need not correspond to a physically realistic real-life outcome. In such cases, the scenario is best viewed as a parameter for encoding an optimal optimal decison.
\end{remark}

\begin{proposition}[Constructive expressions]
\label{prop:constructive}
The optimal scenario components $q_\mu^\#(x,\xi)$, $W_{\mu,\epsilon}^\#(x,\xi)$, and $h_{\mu,\delta}^\#(x)$ have closed form expressions in terms of $z^\#(x)$, which can be evaluated at computational cost at most equivalent to solving a single-scenario problem. See Appendix~\ref{app:realizability-proof}.
\end{proposition}

\paragraph{Analysis of implications.}
The above theorem shows that a scenario generator can realize an optimal decision policy, even under the restriction of generating only one scenario, or even a single component of a single scenario. Decision focused scenario generation is hence equally expressive as direct decision-policy learning.

By Proposition~\ref{prop:constructive}, constructive formulas are available for decision-optimal scenarios in terms of optimal decisions. Of course, one can likewise construct an optimal decision from an optimal scenario by solving a single-scenario problem. Hence, if we know the value of the decision-optimal $q$, $h$, $W$, or $z$, we can infer the value of the others. Learning any of them is hence equivalent in terms of interpretability and realizability.

Additionally, Proposition~\ref{prop:constructive} and Theorem~\ref{thm:realizability} show that the components you learn do not have to even correspond to the structure of the randomness in the stochastic program. For example, nothing prevents using a method for learning $q$, even if $q$ is deterministic in the original problem.

In particular, this solves the issues associated with problems with random recourse $W$. Learning $W$ directly is generally infeasible due to high dimensionality and intractable constraints, but by the above, one can simply apply any method capable of learning $q$ instead. Since there are many machine learning methods available in the literature for learning $q$, but to the best of our knowledge, currently none capable of dealing with instances of randomness in $W$, this resolves a significant bottleneck by extending the applicability of existing methods.

Generally, the choice of learning $q$ or $h$ via decision focused scenario generation, or $z$ directly through decision policy learning, should be made entirely based on the ease of the learning itself, as determined by dimensionality and complexity of the constraint sets of $\Xi^{\mathrm{synthetic}}_q$, $\Xi^{\mathrm{synthetic}}_h$, and $Z$ respectively. See Appendix~\ref{app:dfl-vs-direct} for a more elaborate discussion.

\section{Decision focused learning via log-barrier smoothing}
\label{sec:dfl-logbarrier}
\label{sec:backprop-logbarriers}

For learning decision-optimized scenario policies, deep learning is a natural approach.
To train a neural net with back-propagation, we would need to differentiate the scenario parametrized decision loss
\begin{equation}
\label{eq:scenario-parametrized-erm}
\min_w\ \frac1N\sum_{i=1}^N
\hat G\!\left(\hat z^\#(\phi_w(x_i)),\xi_i\right),
\qquad
\phi_w:\mathcal X\to \Xi^{\mathrm{synthetic}}.
\end{equation}
Formally, for a fixed index $i$, write $\hat\xi_i(w):=\phi_w(x_i)$ and $z_i(w):=\hat z^\#(\hat\xi_i(w))$.
The chain rule gives
\begin{equation}
\label{eq:scenario-chain-rule}
\frac{\partial}{\partial w}\hat G(z_i(w),\xi_i)
=
\frac{\partial \hat G(z_i(w),\xi_i)}{\partial z}
\frac{\partial \hat z^\#(\hat\xi_i(w))}{\partial \hat\xi_i}
\frac{\partial \phi_w(x_i)}{\partial w}.
\end{equation}
The first term $\partial \hat G/\partial z$ can be handled via subgradients due to the convexity of $\hat G$.
The second term presents an issue, since $\hat z^\#$ is generally piecewise constant in $q$.

As it turns out, log-barrier smoothing solves this problem.

\begin{proposition}[Closed form derivatives for log-barriers]
\label{prop:logbarrier-derivatives}
Fix $\mu>0$, $\tilde\mu\geq 0$, and a realized scenario $\tilde\xi\in\Xi$. The scenario parametrized decision loss
\[
\mathcal L_{\mu,\tilde\mu}^{\tilde\xi}(\xi)
:=
\hat G_{\tilde\mu}\!\left(\hat z_\mu^\#(\xi),\tilde\xi\right),
\qquad
\xi=(W,h,q),
\]
is smooth in $q$ and $h$, and $\partial \hat z_\mu^\#/\partial q$ and $\partial \hat z_\mu^\#/\partial h$ have closed form derivatives in terms of $\hat z_\mu^\#(W,h,q)$ and its dual. See Appendix~\ref{app:closed-form-derivatives}.
\end{proposition}

Unfortunately, smoothness is not sufficient to enable learning, since spurious local minima may prevent successful training.
We address this via the following generalized convexity result:

\begin{theorem}[No vanishing gradients]
\label{thm:no-vanishing-gradients}
Fix $x \in \mathcal X$, $\mu > 0$. Choose an arbitrary scenario $\tilde\xi \in \Xi$ and let $\widetilde W = W(\tilde\xi)$ denote its recourse matrix.
The scenario parametrized decision loss $L_\mu(\xi,x):=G_\mu(\hat z_\mu^\#(\xi),x)$ is componentwise invex in $q$ and $h$.
For a pair $(q^\#,h^\#)\in\R^{n_2}\times\R^{m_2}$, the following conditions are equivalent:
\begin{gather}
\nabla_q L_\mu((\widetilde W,h^\#,q^\#),x)=0,
\qquad
\nabla_h L_\mu((\widetilde W,h^\#,q^\#),x)=0
\label{eq:no-vg-stationary}
\\
\Longleftrightarrow
\notag\\
(\widetilde W,h^\#,q^\#)
\in
\argmin_{(W,h,q)\in\Xi^{\mathrm{synthetic}}}
L_\mu((W,h,q),x)
\label{eq:no-vg-loss-min}
\\
\Longleftrightarrow
\notag\\
\hat z_\mu^\#(\widetilde W,h^\#,q^\#)
=
z_\mu^\#(x)
=
\argmin_{z\in\R^{n_1}}G_\mu(z,x)
\label{eq:no-vg-decision-opt}
\end{gather}
\end{theorem}

The theorem implies that the loss landscape over the neural net scenario output induced by log-barrier smoothing is ``trap free'', in the sense that all of its local minima are global minima.
Training may still get stuck in local minima, as is generally the risk in deep learning, but only due to non-convexities in the net itself, not in the loss function.
This gives a principled theoretical justification for log-barrier smoothing, and in particular shows that it avoids the vanishing-gradient pathology known to affect quadratic smoothing.

\section{An interior point inspired method for decision focused scenario generation}
\label{sec:interior-point-method}
\label{sec:nnipm}

The previous section shows that log-barrier smoothing can enable decision focused scenario learning via backpropagation.
But we face a dilemma when setting the smoothing parameter $\mu$.
When $\mu$ is large, gradients will be well-behaved and stable, but the solution of the problem will be distorted.
Classical interior point methods proceed by first solving the problem for a sufficiently large $\mu$, and then progressively decreasing it while tracing the optimal solution via Newton's method.
Below, we present the Neural Net Interior Point Method (NNIPM), applying the same philosophy for decision focused scenario learning.

In an initial stage, we set $\mu$ to a relatively high value, to make training easy and stable.
In a subsequent homotopy stage, we progressively decrease $\mu$ while continuing training, maintaining low training loss as the problem approaches the non-smooth version.
When $\mu$ reaches a sufficiently small target value $\mu_f:=\mu_S$, we begin a final fine-tuning stage where we train directly on the unsmoothed decision loss, solving the partially unsmoothed empirical risk minimization problem
\begin{equation}
\label{eq:nnipm-finetune}
\min_w\
\frac1N\sum_{i=1}^N
\hat G\!\left(\hat z_{\mu_f}^\#(\phi_w(x_i)),\xi_i\right).
\end{equation}
Back-propagation through the unsmoothed loss $\hat G$ is handled via subgradients; see Appendix~\ref{app:closed-form-derivatives}.

\begin{algorithm}[H]
\caption{Neural Net Interior Point Method for scenario generation.}
\label{alg:nnipm-training}
\begin{algorithmic}[1]
\Statex \textbf{Input:} training data $\mathcal D=\{(x_i,\xi_i)\}_{i=1}^N$; scenario generator $\phi_w$; decreasing barrier schedule $\mu_1>\cdots>\mu_S>0$.
\Statex \textbf{Output:} trained parameters $w$.
\State Initialize $w$.
\For{$s=1$ \textbf{to} $S$}
    \State Set $\mu\gets\mu_s$.
    \For{training iterations}
        \State Sample minibatch $\mathcal B\subseteq\mathcal D$.
        \State Generate synthetic scenarios $\hat\xi_i\gets\phi_w(x_i)$ for $(x_i,\xi_i)\in\mathcal B$.
        \State Solve $z_i\gets\hat z_\mu^\#(\hat\xi_i)$ for each generated scenario.
        \State Update $w$ by backpropagating the minibatch loss $\frac1{|\mathcal B|}\sum_{(x_i,\xi_i)\in\mathcal B}\hat G_\mu(z_i,\xi_i)$.
    \EndFor
\EndFor
\For{fine-tuning iterations}
    \State Sample minibatch $\mathcal B\subseteq\mathcal D$.
    \State Generate synthetic scenarios $\hat\xi_i\gets\phi_w(x_i)$ for $(x_i,\xi_i)\in\mathcal B$.
    \State Solve $z_i\gets\hat z_{\mu_S}^\#(\hat\xi_i)$ for each generated scenario.
    \State Update $w$ by backpropagating the true downstream loss $\frac1{|\mathcal B|}\sum_{(x_i,\xi_i)\in\mathcal B}\hat G(z_i,\xi_i)$.
\EndFor
\State \textbf{return} $w$.
\end{algorithmic}
\end{algorithm}

\section{Conclusion}
\label{sec:conclusion}

We have presented both theoretical results, and a novel method. It is important to keep in mind their distinctness. While our theoretical results ground and justify our method, their relevance extends far beyond it, with general relevance for contextual stochastic programming. One could argue that they are the main contribution.

To the best of our knowledge, we are the first to present a comprehensive theory of optimal policy realizability via scenario learning, in the contextual stochastic programming literature. Proposition~\ref{prop:constructive} extends the applicability of various existing methods, connection different methods in the literature, and rendering previously intractable problem instances tractable. Likewise, our theoretical results regarding log-barriers generalize beyond our method and even beyond contextual stochastic programming, providing a principled argument for their usage, and superiority over quadratic smoothing methods, for decision focused learning.

Our method, inspired by these results, has several benefits. It has general applicability, and require no assumptions on the structure of the randomness for the underlying program. It is simultanesously dynamic, and can be adjusted to fit the needs of specific problems in terms of which components it learns. It can be applied both for problems with integer variables and those without. When integer variables are missing or the integrality gap is zero, it has the benefit or training directly on the decision objective, without the need for convex or black-box neural surrogates. On a conceptual level, it functions as a neural net extension of classical interior point methods, which have a rich theory that can be leveraged for theoretical analysis.

However, there are also key limitations to keep in mind. The theoretical results only apply for synthetic scenarios, which function less as realistic predictions and more like arbitrary decision parameters. This limits interpretability. Additionally, the results only apply for problems where the integrality gap is zero. This is a common assumption in the theoretical literature, but still represents a clear weakness.

Regarding our method, it suffers from a limitation common to mathematical programming specialized machine learning methods. While it is applicable to integer problem at inference, it does not take them account during training. For many problems with significant integrality gaps, such as routing problems, we can expect weak performance. Additionally, the method is most relevant for cases where the constraints on at least one scenario component is simple, which is not guaranteed to hold.

Regarding our empirical results, they provide a convincing case for the viability of our method on problems the size of standard literature benchmarks. (They also provide preliminary indication of favorable scalability compared with baselines). However, one should bear in mind that industrial problem instances are often of much larger size.

Our experiments provide preliminary indications of our method scaling favorably compared with baselines. Our experiments provide a convincing case for the viability of our method by considering standard benchmarks, but many realistic industrial problems are of much greater size. Future work will focus on validating the method on such instances.

Future work will focus on extending the theory and the method to take integers into account more thoroughly. Additionally, we expect the theory to generalize both to risk-averse problems, and more general convex programs such as second-order conic programs and semi-definite programs. Experimentally, the next step will be to validate the method on realistic industrial scale problems.

\subsubsection{Realizability and single-scenario optimality}
\label{app:realizability}

\paragraph{Closed form expressions for $W^\#$, $h^\#$, and $q^\#$ in terms of $z^\#$.}
\label{app:realizability-proof}

For a fixed recourse matrix $\widetilde W$, we may use the projection of $\Xi^{\mathrm{synthetic}}$ onto the $(q,h)$ components with recourse matrix $\widetilde W$:
\begin{equation}
\label{eq:appendix-Xi-synthetic-W}
\Xi^{\mathrm{synthetic},\widetilde W}
:=
\left\{
(q,h)\in\R^{n_2}\times\R^{m_2}:\;(\widetilde W,h,q)\in \Xi^{\mathrm{synthetic}}
\right\}.
\end{equation}

Let $x\in\mathcal X$ and $\mu\in[0,\infty)$ be fixed. For compactness, write
\[
\lambda_\mu^{\#,x}:=\lambda_\mu^{x}(z_\mu^\#(x)),
\qquad
\tilde\lambda_\mu^{x}:=\hat\lambda_\mu(z_\mu^\#(x),\tilde\xi),
\]
with $\mu=0$ interpreted by the central-path limit.

\paragraph{Learning $q$.}
For any $\tilde\xi\in\Xi$, the decision-optimal cost vector in Theorem~\ref{thm:realizability} is
\begin{equation}
\label{eq:q-construct}
q_\mu^{\#,\tilde\xi}(x)
:=
\tilde q+\tilde W^\top\!\left(\lambda_\mu^{\#,x}-\tilde\lambda_\mu^{x}\right).
\end{equation}
Thus the synthetic scenario is
\begin{equation}
\label{eq:q-scenario-appendix}
\xi_\mu^{\#,x,\tilde\xi}
:=
\bigl(\tilde W,\tilde h,q_\mu^{\#,\tilde\xi}(x)\bigr).
\end{equation}

\paragraph{Learning $W$.}
Assuming the componentwise ratios below are well-defined and positive, the equivalent decision-optimal recourse matrix is
\begin{equation}
\label{eq:W-construct}
W_\mu^{\#,\tilde\xi}(x)
:=
\tilde W\,
\diag\!\left(
\frac{\tilde q}{q_\mu^{\#,\tilde\xi}(x)}
\right),
\end{equation}
where division is componentwise. The corresponding synthetic scenario is
\begin{equation}
\label{eq:W-scenario-appendix}
\xi_\mu^{\#,x,\tilde\xi}
:=
\bigl(W_\mu^{\#,\tilde\xi}(x),\tilde h,\tilde q\bigr).
\end{equation}
Indeed, the change of variables
\[
y
=
\diag\!\left(
\frac{\tilde q}{q_\mu^{\#,\tilde\xi}(x)}
\right)\tilde y
\]
maps the second-stage problem with data $(\tilde W,\tilde h,q_\mu^{\#,\tilde\xi}(x))$ to the second-stage problem with data $(W_\mu^{\#,\tilde\xi}(x),\tilde h,\tilde q)$, up to the additive log-barrier constant when $\mu>0$.

\paragraph{Learning $h$.}
When $W(\xi)=W$ for all $\xi\in\Xi$, the decision-optimal right-hand side is
\begin{equation}
\label{eq:h-construct}
h_\mu^{\#,\delta}(x):=
Tz_\mu^\#(x)+W y_\mu^{\#,\delta}(x),
\qquad
y_\mu^{\#,\delta}(x):=
\begin{cases}
\mu\left(q^{\max,\delta}-W^\top\lambda_\mu^{\#,x}\right)^{-1}, & \mu>0,\\
0, & \mu=0.
\end{cases}
\end{equation}
The corresponding synthetic scenario is
\begin{equation}
\label{eq:h-scenario-appendix}
\xi_\mu^{\#,x,\delta}
:=
\bigl(W,h_\mu^{\#,\delta}(x),q^{\max,\delta}\bigr).
\end{equation}

\paragraph{Closed form expressions for derivatives of $G$ and the optimization map.}
\label{app:closed-form-derivatives}

Fix $\mu>0$ and a synthetic scenario $\xi=(W,h,q)$ for which the single-scenario smoothed problem is well defined. Let
\[
\hat z_\mu^\#:=\hat z_\mu^\#(\xi),
\qquad
\hat\lambda_\mu^\#:=\hat\lambda_\mu(\hat z_\mu^\#,\xi)
\]
denote the scenario-specific solution and its associated second-stage dual multiplier. For $z$ in the relative interior of $Z$,
\[
\partial_z G_\mu(z,x)
=
c-\mu z^{-1}
-T^\top\E_{\boldsymbol{\xi}\sim P(\cdot\mid x)}
\!\left[\hat\lambda_\mu(z,\boldsymbol{\xi})\right]
+N_Z(z),
\]
with the scenario-specific expression
\[
\partial_z \hat G_\mu(z,\xi)
=
c-\mu z^{-1}-T^\top\hat\lambda_\mu(z,\xi)+N_Z(z).
\]
Whenever the displayed inverses exist,
\[
\begin{aligned}
\frac{\partial \hat z_\mu^\#(\xi)}{\partial h}
&=
P_\mu(\xi)T^\top S_\mu(\xi)^{-1},
\\
\frac{\partial \hat z_\mu^\#(\xi)}{\partial q}
&=
P_\mu(\xi)T^\top S_\mu(\xi)^{-1}\mu W D_\mu(\xi),
\end{aligned}
\]
where
\[
\begin{aligned}
H_\mu(\xi)&:=\diag\!\big((\hat z_\mu^\#)^2\big),\\
P_\mu(\xi)&:=\frac1\mu\left(H_\mu(\xi)-H_\mu(\xi)A^\top(AH_\mu(\xi)A^\top)^{-1}AH_\mu(\xi)\right),\\
D_\mu(\xi)&:=\diag\!\left((q-W^\top\hat\lambda_\mu^\#)^{-2}\right),\\
S_\mu(\xi)&:=TP_\mu(\xi)T^\top+\mu W D_\mu(\xi)W^\top.
\end{aligned}
\]
